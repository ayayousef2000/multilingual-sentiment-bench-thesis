\chapter*{СПИСОК СОКРАЩЕНИЙ И УСЛОВНЫХ ОБОЗНАЧЕНИЙ}
\addcontentsline{toc}{chapter}{СПИСОК СОКРАЩЕНИЙ И УСЛОВНЫХ ОБОЗНАЧЕНИЙ}

\noindent CNN – Convolutional Neural Network, сверточная нейронная сеть

\noindent DAG – Directed Acyclic Graph, ориентированный ациклический граф

\noindent DOM – Document Object Model, объектная модель документа

\noindent FIFO – First In, First Out, принцип последовательной обработки запросов

\noindent FP8 / FP16 / FP32 – форматы чисел с плавающей точкой (8/16/32 бит)

\noindent GDPR – General Data Protection Regulation, общий регламент защиты персональных данных

\noindent GELU – Gaussian Error Linear Unit, функция активации на основе гауссовой ошибки

\noindent GPU – Graphics Processing Unit, графический процессор

\noindent GRU – Gated Recurrent Unit, управляемый рекуррентный блок

\noindent INT4 / INT8 – целочисленные форматы квантования (4-бит / 8-бит)

\noindent JIT – Just-In-Time, динамическая компиляция

\noindent LSTM – Long Short-Term Memory, долгая краткосрочная память

\noindent MLM – Masked Language Modeling, маскированное языковое моделирование

\noindent NLP – Natural Language Processing, обработка естественного языка

\noindent NSP – Next Sentence Prediction, предсказание следующего предложения

\noindent ONNX – Open Neural Network Exchange, открытый формат обмена нейронными сетями

\noindent RTT – Round-Trip Time, время двунаправленной передачи данных по сети

\noindent SIMD – Single Instruction Multiple Data, параллельная обработка данных одной инструкцией

\noindent SPA – Single Page Application, одностраничное веб-приложение

\noindent SVM – Support Vector Machine, метод опорных векторов

\noindent UI – User Interface, пользовательский интерфейс

\noindent WASM – WebAssembly, бинарный формат инструкций для браузерной виртуальной машины

\noindent WebGPU – Web Graphics Processing Unit API, стандарт графического API для браузера
